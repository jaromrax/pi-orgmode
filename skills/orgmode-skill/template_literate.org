#+begin_src python
import time
import threading

canvas=ROOT.TCanvas("a")
def run_root_loop(canvas):
    while not canvas.IsZombie():
        ROOT.gSystem.ProcessEvents()
        time.sleep(0.05)
def show_canvas(): # Use to let ROOT show the Draw
    thread = threading.Thread(target=run_root_loop, args=(canvas,), daemon=True)
    thread.start()
# -------------------------------------------------
h=ROOT.TH2F("h","h",100,0,100,100,0,100)
canvas.Draw()
h.Draw()
show_canvas()

**** Using HEADERS:
 - Org can extract one or more source code blocks and write them to one or more source files—a process known as tangling in literate programming terminology (:tangle yes).
 - it has the ability to pass variables, functions, and results to one another (:noweb yes)

Defaults:
 - :session    => "none"  \quad /buffer for debug/
 - :results    => "replace" \quad  /replace output... try also file raw drawer/
   - /drawer/ makes a kind of drawer...
   - /replace file output/ prepares image display if the only print is imagename
   - you may need ~M-x iimage-mode~, maybe not
 - :exports    => "code"\quad  /both code results/
 - :cache      => "no" \quad  /not tested, better avoid/
 - :noweb      => "no" \quad  ~/allow using <<name>> to call prev block/~
 - :hlines     => "no" \quad  / /
 - :tangle     => "no" \quad  /save code to file, all to same file; try FILENAME/
 - :comments   => "yes" \quad /insert comments to tanglefile, lot of info/
